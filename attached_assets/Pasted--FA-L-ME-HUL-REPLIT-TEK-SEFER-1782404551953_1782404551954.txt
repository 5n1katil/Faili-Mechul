import { useMemo, useState } from "react";

const normalizeRotation = (value) => ((Number(value ?? 0) % 360) + 360) % 360;

export default function BrokenCompassCalibrationMiniGame({ clue, onSolved, onRequestHint }) {
  const sifre = clue?.sifre ?? {};
  const p = sifre.presentation ?? {};
  const segments = Array.isArray(p.segments) ? p.segments : [];
  const slots = Array.isArray(p.slots) ? p.slots : [];
  const correctPlacement = Array.isArray(p.correctPlacement) ? p.correctPlacement : [];
  const segmentById = useMemo(() => new Map(segments.map((segment) => [segment.id, segment])), [segments]);
  const initialRotations = useMemo(() => Object.fromEntries(segments.map((segment) => [segment.id, normalizeRotation(segment.initialRotation)])), [segments]);
  const [order, setOrder] = useState([]);
  const [rotations, setRotations] = useState(initialRotations);
  const [result, setResult] = useState(null);
  const [hintOpen, setHintOpen] = useState(false);
  const [hintUsed, setHintUsed] = useState(false);
  const [solved, setSolved] = useState(false);
  const interaction = p.interaction ?? {};
  const valid = segments.length >= 5 && slots.length === 4 && correctPlacement.length === slots.length;

  const CompassFace = ({ segment, rotation }) => {
    const marks = Array.isArray(segment.marks) ? segment.marks : [];
    const at = (index) => marks[index] ?? "·";
    const face = { width:54, height:54, border:"1px solid " + (segment.metal === "krom" ? "#aebfd8" : "#b99b57"), borderRadius:"50%", position:"relative", margin:"0 auto", color:segment.metal === "krom" ? "#e2eafa" : "#f6e9bf", background:segment.metal === "krom" ? "radial-gradient(circle,#394656,#151923 68%)" : "radial-gradient(circle,#40331b,#17150d 68%)", transform:"rotate(" + rotation + "deg)" };
    const mark = { position:"absolute", fontSize:9, fontWeight:800, lineHeight:1 };
    return <div style={face}><span style={{ ...mark, top:4, left:"50%", transform:"translateX(-50%)" }}>{at(0)}</span><span style={{ ...mark, right:4, top:"50%", transform:"translateY(-50%)" }}>{at(1)}</span><span style={{ ...mark, bottom:4, left:"50%", transform:"translateX(-50%)" }}>{at(2)}</span><span style={{ ...mark, left:4, top:"50%", transform:"translateY(-50%)" }}>{at(3)}</span><span style={{ position:"absolute", inset:0, display:"grid", placeItems:"center", fontFamily:"serif", fontSize:17 }}>{segment.innerRune ?? "◌"}</span></div>;
  };
  const toggle = (segmentId) => { setResult(null); setOrder((before) => before.includes(segmentId) ? before.filter((id) => id !== segmentId) : (before.length < slots.length ? [...before, segmentId] : before)); };
  const rotate = (segmentId, delta) => { setResult(null); setRotations((before) => ({ ...before, [segmentId]: normalizeRotation((before[segmentId] ?? 0) + delta) })); };
  const reset = () => { setOrder([]); setRotations(initialRotations); setResult(null); setSolved(false); };
  const askHint = () => { if (!hintUsed) { onRequestHint?.({ clueId: clue.id, penaltySeconds: Number(sifre.hintPenaltySeconds ?? 60) }); setHintUsed(true); } setHintOpen(true); };
  const submit = () => {
    const ok = order.length === slots.length && correctPlacement.every((need, index) => need.slotId === slots[index]?.id && need.segmentId === order[index] && normalizeRotation(rotations[need.segmentId]) === normalizeRotation(need.rotation));
    setResult({ ok, message: ok ? (interaction.successMessage ?? "İç halka kapandı.") : (interaction.failureMessage ?? "Halka henüz kapanmadı.") });
    if (ok && !solved) { setSolved(true); onSolved?.({ clueId: clue.id, mechanic: "broken_compass_calibration" }); }
  };
  if (!valid) return <p role="alert">Bu vaka için kırık pusula kalibrasyon verisi eksik veya tutarsız.</p>;
  const shell = { background:"#101510", border:"1px solid #5e6044", borderRadius:12, padding:14, color:"#e8eaf2" };
  const fragmentBase = { minHeight:108, border:"1px solid #655d45", borderRadius:10, padding:8, textAlign:"center", cursor:"pointer", color:"#f6e9bf" };

  return (
    <section style={shell} aria-label={p.accessibility?.description ?? "Kırık pusula kalibrasyonu"}>
      <h4 style={{ margin:"0 0 6px", color:"#f0a830" }}>🧭 {p.title ?? "Kırık Pusula Kalibrasyonu"}</h4>
      <p style={{ margin:"0 0 10px", color:"#8b91ad", fontSize:13 }}>{p.subtitle ?? "Parçaların çentiklerini halka boyunca eşleştir."}</p>
      {p.purposeHint ? <p style={{ margin:"0 0 10px", padding:10, borderLeft:"2px solid #e0b54e", background:"rgba(224,181,78,.06)", color:"#ecd99a", fontSize:13 }}><b>ÇÖZÜMÜN İŞLEVİ</b><br />{p.purposeHint}</p> : null}
      <p style={{ margin:"0 0 10px", padding:9, border:"1px solid #4e563f", borderRadius:8, color:"#d9e7c5", fontSize:12 }}><b>{p.sceneLabel ?? "SEYRÜSEFER KAYDI"}</b><br />{p.northKey}</p>
      <div style={{ display:"grid", gridTemplateColumns:"repeat(auto-fit,minmax(120px,1fr))", gap:8 }}>
        {segments.map((segment) => { const pos=order.indexOf(segment.id); const rotation=normalizeRotation(rotations[segment.id]); return <button key={segment.id} type="button" onClick={() => toggle(segment.id)} aria-pressed={pos>=0} style={{ ...fragmentBase, borderColor:pos>=0 ? "#22c55e" : (segment.metal === "krom" ? "#6e7c92" : "#655d45"), background:pos>=0 ? "#213426" : (segment.metal === "krom" ? "#1b2230" : "#30291d") }}><span style={{ display:"block", fontSize:10, color:segment.metal === "krom" ? "#c7d5ec" : "#cdbb79", marginBottom:5 }}>{segment.label} · {segment.metal}</span><span style={{ display:"flex", justifyContent:"space-between", gap:6, color:segment.metal === "krom" ? "#c7d5ec" : "#d1bd80", fontSize:10, marginBottom:5 }}><span>◁ {segment.leftNotch ?? "?"}</span><span>{segment.rightNotch ?? "?"} ▷</span></span><CompassFace segment={segment} rotation={rotation} />{pos>=0 ? <span style={{ display:"block", marginTop:6, fontSize:10, color:"#86efac" }}>Yuva {pos + 1}</span> : null}</button>; })}
      </div>
      <div style={{ display:"grid", gridTemplateColumns:"repeat(auto-fit,minmax(135px,1fr))", gap:8, marginTop:12 }}>
        {slots.map((slot,index) => { const segment=segmentById.get(order[index]); if(!segment) return <div key={slot.id} style={{ minHeight:120, border:"1px dashed #655d45", borderRadius:10, padding:9, textAlign:"center", color:"#8b91ad" }}><b style={{ color:"#cdbb79", fontSize:10 }}>{slot.label} · {slot.requiredMark}</b><p style={{ fontSize:11 }}>Bu yuvaya bir kadran yerleştir.</p></div>; const rotation=normalizeRotation(rotations[segment.id]); return <div key={slot.id} style={{ minHeight:140, border:"1px solid #655d45", borderRadius:10, padding:9, textAlign:"center", background:"#11150f" }}><b style={{ color:"#cdbb79", fontSize:10 }}>{slot.label} · {slot.requiredMark}</b><div style={{ display:"flex", justifyContent:"space-between", gap:6, color:segment.metal === "krom" ? "#c7d5ec" : "#d1bd80", fontSize:10, margin:"6px 0" }}><span>◁ {segment.leftNotch ?? "?"}</span><span>{segment.rightNotch ?? "?"} ▷</span></div><CompassFace segment={segment} rotation={rotation} /><div style={{ fontSize:10, color:"#8b91ad", margin:"6px 0" }}>{segment.label} · {rotation}°</div><div style={{ display:"flex", justifyContent:"center", gap:6 }}><button type="button" onClick={() => rotate(segment.id,-90)}>↶ 90°</button><button type="button" onClick={() => rotate(segment.id,90)}>↷ 90°</button></div></div>; })}
      </div>
      <div style={{ display:"flex", gap:8, flexWrap:"wrap", marginTop:10 }}><button type="button" onClick={reset} style={{ flex:1, padding:10, borderRadius:8, border:"1px solid #262c44", background:"#1c2138", color:"#e8eaf2" }}>{interaction.resetLabel ?? "Kalibrasyonu Temizle"}</button><button type="button" onClick={submit} style={{ flex:1, padding:10, border:0, borderRadius:8, background:"#e0b54e", color:"#1a1205", fontWeight:700 }}>{interaction.submitLabel ?? "Pusulayı Kalibre Et"}</button></div>
      <button type="button" onClick={askHint} disabled={hintUsed} style={{ width:"100%", marginTop:10, padding:10, borderRadius:8, border:"1px dashed #b45309", background:"transparent", color:"#fbbf24" }}>{hintUsed ? "İpucu açıldı" : "İpucu iste (-" + Number(sifre.hintPenaltySeconds ?? 60) + " sn)"}</button>
      {hintOpen ? <p style={{ margin:"8px 0 0", padding:9, borderLeft:"2px solid #b45309", background:"rgba(180,83,9,.08)", color:"#fbbf24", fontSize:13 }}>{sifre.cozumIpucu}</p> : null}
      {result ? <p role="status" style={{ margin:"10px 0 0", padding:10, borderRadius:8, background:result.ok ? "rgba(34,197,94,.12)" : "rgba(220,38,38,.12)", color:result.ok ? "#86efac" : "#fca5a5" }}>{result.ok ? "✓ " : "✕ "}{result.message}</p> : null}
      {result?.ok ? <><p style={{ margin:"10px 0 0", padding:10, border:"1px solid #22c55e", borderRadius:8, background:"rgba(34,197,94,.08)", color:"#b7f7c8" }}><b>OKUNAN ROTA NOTU</b><br />{p.routeResult}</p>{sifre.aciklama ? <p style={{ margin:"10px 0 0", color:"#8b91ad", fontStyle:"italic", fontSize:13 }}>{sifre.aciklama}</p> : null}</> : null}
    </section>
  );
}
