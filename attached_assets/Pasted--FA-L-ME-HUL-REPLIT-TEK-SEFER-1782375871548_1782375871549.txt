// src/components/caseMechanics/SymbolCrossgridMiniGame.jsx
import { useMemo, useState } from "react";

const normalizeAnswer = (value) =>
  String(value ?? "")
    .toLocaleLowerCase("tr-TR")
    .replace(/[^\p{L}\p{N}]/gu, "")
    .trim();

export default function SymbolCrossgridMiniGame({ clue, onSolved, onRequestHint }) {
  const [answer, setAnswer] = useState("");
  const [result, setResult] = useState(null);
  const [hintOpen, setHintOpen] = useState(false);
  const [hintUsed, setHintUsed] = useState(false);

  const sifre = clue?.sifre ?? {};
  const presentation = sifre.presentation ?? {};
  const rows = Array.isArray(presentation.rowSymbols) ? presentation.rowSymbols : [];
  const columns = Array.isArray(presentation.columnSymbols) ? presentation.columnSymbols : [];
  const cells = Array.isArray(presentation.cells) ? presentation.cells : [];
  const validBoard =
    rows.length > 0 &&
    rows.length === columns.length &&
    cells.length === rows.length &&
    cells.every((row) => Array.isArray(row) && row.length === columns.length);

  const acceptedAnswers = useMemo(() => {
    const aliases = [
      sifre.cozulmus,
      ...(Array.isArray(sifre.answerAliases) ? sifre.answerAliases : []),
      ...(Array.isArray(presentation.answerAliases) ? presentation.answerAliases : []),
    ];
    return aliases.map(normalizeAnswer).filter(Boolean);
  }, [sifre, presentation]);

  const interaction = presentation.interaction ?? {};
  const purposeHint = presentation.purposeHint ?? presentation.objectiveHint ?? "";
  const hintPenaltySeconds = Number(sifre.hintPenaltySeconds ?? 60);
  const minCell = Math.max(36, Number(presentation.mobile?.minCellPx ?? 42));

  const submit = () => {
    const correct = acceptedAnswers.includes(normalizeAnswer(answer));
    if (correct) {
      setResult({ ok: true, message: interaction.successMessage ?? "Şifre çözüldü." });
      onSolved?.({ clueId: clue.id, mechanic: "symbol_crossgrid", answer: answer.trim() });
      return;
    }
    setResult({ ok: false, message: interaction.failureMessage ?? "Henüz değil. Aynı sembolün satırı ile sütununu eşleştir." });
  };

  const openHint = () => {
    if (!hintUsed) {
      onRequestHint?.({ clueId: clue.id, penaltySeconds: hintPenaltySeconds });
      setHintUsed(true);
    }
    setHintOpen(true);
  };

  if (!validBoard) {
    return <div role="alert">Bu vaka için sembol eşleme tahtası eksik veya kare değil.</div>;
  }

  const shell = { background: "#10131f", border: "1px solid #262c44", borderRadius: 12, padding: 14, color: "#e8eaf2" };
  const cell = { width: minCell, minWidth: minCell, height: minCell, border: "1px solid #262c44", textAlign: "center", fontWeight: 700 };
  const header = { ...cell, position: "sticky", top: 0, zIndex: 2, background: "#1c2138", color: "#e0b54e" };
  const rowHeader = { ...cell, position: "sticky", left: 0, zIndex: 1, background: "#171c2e", color: "#f0a830" };
  const corner = { ...header, left: 0, zIndex: 3, background: "#14192a" };

  return (
    <section style={shell} aria-label={presentation.accessibility?.description ?? "Sembol eşleme tahtası"}>
      <h4 style={{ margin: "0 0 6px", color: "#f0a830" }}>{presentation.title ?? "Eşleme Tahtası"}</h4>
      <p style={{ margin: "0 0 10px", color: "#8b91ad", fontSize: 13 }}>{presentation.subtitle ?? "Satır ve sütunda aynı sembolü eşleştir."}</p>
      {purposeHint ? <p style={{ margin: "0 0 12px", padding: 10, borderLeft: "2px solid #e0b54e", background: "rgba(224,181,78,.06)", color: "#ecd99a", fontSize: 13 }}><b>ÇÖZÜMÜN İŞLEVİ</b><br />{purposeHint}</p> : null}
      <div style={{ overflowX: "auto", border: "1px solid #262c44", borderRadius: 10, padding: 8 }}>
        <table style={{ borderCollapse: "separate", borderSpacing: 0, margin: "0 auto" }}>
          <thead>
            <tr>
              <th style={corner}>↘</th>
              {columns.map((symbol) => <th key={symbol} scope="col" style={header}>{symbol}</th>)}
            </tr>
          </thead>
          <tbody>
            {rows.map((symbol, rowIndex) => (
              <tr key={symbol}>
                <th scope="row" style={rowHeader}>{symbol}</th>
                {columns.map((columnSymbol, columnIndex) => (
                  <td key={columnSymbol} style={{ ...cell, background: "#161a2b", color: "#dbe2f7" }}>{cells[rowIndex][columnIndex]}</td>
                ))}
              </tr>
            ))}
          </tbody>
        </table>
      </div>
      <div style={{ marginTop: 12, padding: 10, borderRadius: 8, background: "#161a2b", color: "#e0b54e", letterSpacing: 2, textAlign: "center" }}>{presentation.cipherDisplay ?? (presentation.cipherSymbols ?? []).join(" ")}</div>
      <button type="button" onClick={openHint} disabled={hintUsed} style={{ width: "100%", marginTop: 10, padding: 10, borderRadius: 8, border: "1px dashed #b45309", background: "transparent", color: "#fbbf24" }}>{hintUsed ? "İpucu açıldı" : "İpucu iste (-" + hintPenaltySeconds + " sn)"}</button>
      {hintOpen ? <p style={{ margin: "8px 0 0", padding: 9, borderLeft: "2px solid #b45309", background: "rgba(180,83,9,.08)", color: "#fbbf24", fontSize: 13 }}>{sifre.cozumIpucu}</p> : null}
      <input value={answer} onChange={(event) => setAnswer(event.target.value)} onKeyDown={(event) => event.key === "Enter" && submit()} aria-label={interaction.inputLabel ?? "Çözülmüş metni yaz"} placeholder={interaction.inputLabel ?? "Çözülmüş metni yaz..."} style={{ width: "100%", boxSizing: "border-box", marginTop: 10, padding: 11, borderRadius: 8, border: "1px solid #262c44", background: "#161a2b", color: "#e8eaf2", textAlign: "center" }} />
      <button type="button" onClick={submit} style={{ width: "100%", marginTop: 10, padding: 11, border: 0, borderRadius: 8, background: "#e0b54e", color: "#1a1205", fontWeight: 700 }}>{interaction.submitLabel ?? "Şifreyi Çöz"}</button>
      {result ? <p role="status" style={{ margin: "10px 0 0", padding: 10, borderRadius: 8, background: result.ok ? "rgba(34,197,94,.12)" : "rgba(220,38,38,.12)", color: result.ok ? "#86efac" : "#fca5a5" }}>{result.ok ? "✓ " : "✕ "}{result.message}</p> : null}
      {result?.ok && sifre.aciklama ? <p style={{ margin: "10px 0 0", color: "#8b91ad", fontStyle: "italic" }}>{sifre.aciklama}</p> : null}
    </section>
  );
}
