// src/components/caseMechanics/TelephoneSwitchboardMiniGame.jsx
import { useRef, useState } from "react";

export default function TelephoneSwitchboardMiniGame({ clue, onSolved, onRequestHint }) {
  const puzzle = clue?.audioPuzzle ?? {};
  const audioRef = useRef(null);
  const stopAtRef = useRef(null);
  const [selectedId, setSelectedId] = useState(null);
  const [result, setResult] = useState(null);
  const [hintOpen, setHintOpen] = useState(false);
  const [transcriptOpen, setTranscriptOpen] = useState(false);
  const [activeSegmentId, setActiveSegmentId] = useState(null);

  const segments = Array.isArray(puzzle.segments) ? puzzle.segments : [];
  const options = Array.isArray(puzzle.options) ? puzzle.options : [];
  const interaction = puzzle.interaction ?? {};
  const hintPenaltySeconds = Number(puzzle.hintPenaltySeconds ?? 60);

  const playSegment = (segment) => {
    const audio = audioRef.current;
    if (!audio || !segment) return;
    stopAtRef.current = Number(segment.endSec);
    audio.currentTime = Math.max(0, Number(segment.startSec ?? 0));
    setActiveSegmentId(segment.id);
    audio.play().catch(() => {});
  };

  const stopAtBoundary = () => {
    const audio = audioRef.current;
    if (audio && Number.isFinite(stopAtRef.current) && audio.currentTime >= stopAtRef.current) {
      audio.pause();
      stopAtRef.current = null;
      setActiveSegmentId(null);
    }
  };

  const chooseLine = (optionId) => {
    setSelectedId(optionId);
    const ok = optionId === puzzle.correctOptionId;
    setResult({ ok, message: ok ? (puzzle.successMessage ?? "Hat, ızgaradaki bağlantıyı kesinleştirdi.") : (puzzle.failureMessage ?? "Kesitleri yeniden dinle.") });
    if (ok) onSolved?.({ clueId: clue.id, mechanic: "telephone_switchboard" });
  };

  const askHint = () => {
    if (!hintOpen) onRequestHint?.({ clueId: clue.id, penaltySeconds: hintPenaltySeconds });
    setHintOpen(true);
  };

  return (
    <section aria-label={puzzle.title ?? "Dahili Hat Düzeni"} style={{ marginTop: 12, padding: 14, border: "1px solid #262c44", borderRadius: 12, background: "#0e1119" }}>
      <h4 style={{ margin: "0 0 8px", color: "#f0a830", fontSize: 13 }}>{puzzle.title ?? "Dahili Hat Düzeni"}</h4>
      {puzzle.subtitle ? <p style={{ margin: "0 0 10px", color: "#8b91ad", fontSize: 13 }}>{puzzle.subtitle}</p> : null}
      {puzzle.purposeHint ? <p style={{ margin: "0 0 10px", padding: 10, borderLeft: "2px solid #e0b54e", background: "rgba(224,181,78,.06)", color: "#ecd99a", fontSize: 13 }}><b>ÇÖZÜMÜN İŞLEVİ</b><br />{puzzle.purposeHint}</p> : null}
      {clue.audioUrl ? <audio ref={audioRef} controls preload="metadata" src={clue.audioUrl} onTimeUpdate={stopAtBoundary} onEnded={() => { stopAtRef.current = null; setActiveSegmentId(null); }} style={{ width: "100%" }}>Tarayıcınız ses oynatmayı desteklemiyor.</audio> : <p style={{ color: "#fbbf24" }}>Ses dosyası yolu tanımlı değil; transcript erişilebilir olmalı.</p>}
      <p style={{ color: "#f0a830", fontSize: 11, letterSpacing: 1, margin: "12px 0 6px" }}>MAKARADAN KESİT DİNLE</p>
      <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fit,minmax(120px,1fr))", gap: 8 }}>
        {segments.map((segment) => <button key={segment.id} type="button" onClick={() => playSegment(segment)} style={{ padding: 10, borderRadius: 8, border: activeSegmentId === segment.id ? "1px solid #f0a830" : "1px solid #262c44", background: "#161a2b", color: "#e8eaf2" }}>▶ {segment.label ?? segment.id}</button>)}
      </div>
      <p style={{ margin: "14px 0 8px", color: "#d6dae8", fontSize: 13 }}>{puzzle.question ?? "Doğru hattı seç."}</p>
      <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fit,minmax(100px,1fr))", gap: 8 }}>
        {options.map((option) => {
          const isSelected = selectedId === option.id;
          const correct = isSelected && option.id === puzzle.correctOptionId;
          return <button key={option.id} type="button" onClick={() => chooseLine(option.id)} style={{ padding: 10, borderRadius: 8, border: correct ? "1px solid #22c55e" : isSelected ? "1px solid #dc2626" : "1px solid #262c44", background: correct ? "rgba(34,197,94,.12)" : isSelected ? "rgba(220,38,38,.12)" : "#161a2b", color: "#e8eaf2" }}>{option.label ?? option.id}</button>;
        })}
      </div>
      {puzzle.hint ? <><button type="button" disabled={hintOpen} onClick={askHint} style={{ width: "100%", marginTop: 10, padding: 10, borderRadius: 8, border: "1px dashed #b45309", background: "transparent", color: "#fbbf24" }}>{hintOpen ? "İpucu açıldı" : "İpucu iste (-" + hintPenaltySeconds + " sn)"}</button>{hintOpen ? <p style={{ margin: "8px 0 0", padding: 9, borderLeft: "2px solid #b45309", color: "#fbbf24", background: "rgba(180,83,9,.08)", fontSize: 13 }}>{puzzle.hint}</p> : null}</> : null}
      {clue.sesMetni && interaction.showTranscriptOnDemand !== false ? <><button type="button" onClick={() => setTranscriptOpen((value) => !value)} style={{ width: "100%", marginTop: 8, padding: 10, borderRadius: 8, border: "1px solid #262c44", background: "#161a2b", color: "#e8eaf2" }}>{transcriptOpen ? "Kayıt çözümlemesini gizle" : "Kayıt çözümlemesini göster"}</button>{transcriptOpen ? <p style={{ margin: "8px 0 0", padding: 10, borderRadius: 8, background: "rgba(224,181,78,.06)", color: "#ecd99a", fontSize: 13 }}>{clue.sesMetni}</p> : null}</> : null}
      {result ? <p role="status" style={{ margin: "10px 0 0", padding: 10, borderRadius: 8, background: result.ok ? "rgba(34,197,94,.12)" : "rgba(220,38,38,.12)", color: result.ok ? "#86efac" : "#fca5a5" }}>{result.ok ? "✓ " : "✕ "}{result.message}</p> : null}
    </section>
  );
}
