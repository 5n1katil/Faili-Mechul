import { useMemo, useState } from "react";

const premiumStyles = new Set([
  "luggage_label_match",
  "negative_contact_sheet",
  "morse_transceiver",
  "archive_index_reconstruction",
  "mechanical_lock_sequence",
]);

export const isPremiumMechanic = (clue) =>
  clue?.mechanicType === "sifreli_mesaj" &&
  premiumStyles.has(clue?.sifre?.presentation?.style);

export default function PremiumMechanicRenderer({ clue, onSolved, onRequestHint }) {
  const p = clue?.sifre?.presentation ?? {};
  const style = p.style;
  const [choices, setChoices] = useState({});
  const [dials, setDials] = useState(() =>
    Object.fromEntries((p.dials ?? []).map((d) => [d.id, d.initial ?? d.symbols?.[0] ?? ""]))
  );
  const [result, setResult] = useState(null);
  const [hintOpen, setHintOpen] = useState(false);
  const [hintUsed, setHintUsed] = useState(false);

  const setChoice = (itemId, answerId) =>
    setChoices((old) => ({ ...old, [itemId]: answerId }));

  const rotate = (dial, delta) => {
    const symbols = dial.symbols ?? [];
    const current = dials[dial.id] ?? symbols[0];
    const index = Math.max(0, symbols.indexOf(current));
    const next = symbols[(index + delta + symbols.length) % symbols.length];
    setDials((old) => ({ ...old, [dial.id]: next }));
    setResult(null);
  };

  const valid = useMemo(() => {
    if (!premiumStyles.has(style)) return false;
    if (style === "mechanical_lock_sequence") {
      return (p.correctCode ?? []).length === (p.dials ?? []).length &&
        (p.correctCode ?? []).every((entry) => dials[entry.dialId] === entry.symbol);
    }
    return (p.correctAssignments ?? []).length > 0 &&
      (p.correctAssignments ?? []).every((entry) => choices[entry.itemId] === entry.answerId);
  }, [choices, dials, p, style]);

  const submit = () => {
    if (valid) {
      setResult("ok");
      onSolved?.({ clueId: clue.id, mechanic: style });
    } else setResult("no");
  };

  const useHint = () => {
    setHintOpen(true);
    if (!hintUsed) {
      setHintUsed(true);
      onRequestHint?.({ clueId: clue.id, penaltySeconds: 60 });
    }
  };

  const select = (itemId, options, placeholder) => (
    <select value={choices[itemId] ?? ""} onChange={(e) => { setChoice(itemId, e.target.value); setResult(null); }}>
      <option value="">{placeholder}</option>
      {options.map((option) => <option key={option.id ?? option} value={option.id ?? option}>{option.label ?? option}</option>)}
    </select>
  );

  const card = (title, body) => <article style={styles.card}><b>{title}</b>{body}</article>;
  let board = null;

  if (style === "luggage_label_match") {
    board = <div style={styles.grid}>{(p.cases ?? []).map((item) => card(item.label, <><small>{item.stamp} · {item.note}</small>{select(item.id, p.routes ?? [], "Rotayı seç…")}</>))}</div>;
  } else if (style === "negative_contact_sheet") {
    board = <div style={styles.grid}>{(p.checks ?? []).map((item) => card(item.label, <><div style={styles.pair}><div style={styles.positive}>{item.positive}</div><div style={styles.negative}>{item.negative}</div></div>{select(item.id, item.options ?? [], "Farkı seç…")}</>))}</div>;
  } else if (style === "morse_transceiver") {
    board = <><div style={styles.chart}>{(p.chart ?? []).map((item) => <span key={item.char}><b>{item.char}</b> {item.code}</span>)}</div><div style={styles.grid}>{(p.signals ?? []).map((item) => card(item.label, <><div style={styles.morse}>{item.code}</div>{select(item.id, item.options ?? [], "Harfi seç…")}</>))}</div></>;
  } else if (style === "archive_index_reconstruction") {
    board = <div style={styles.archive}>{(p.files ?? []).map((item) => card(item.code ?? item.label, <><small>{item.date}<br />{item.note}</small>{select(item.id, p.slots ?? [], "Raf gözü…")}</>))}</div>;
  } else if (style === "mechanical_lock_sequence") {
    board = <div style={styles.dials}>{(p.dials ?? []).map((dial) => <article key={dial.id} style={styles.dial}><b>{dial.label}</b><div style={styles.symbol}>{dials[dial.id]}</div><div style={styles.controls}><button onClick={() => rotate(dial, -1)}>↶</button><button onClick={() => rotate(dial, 1)}>↷</button></div></article>)}</div>;
  }

  return <section style={styles.wrap}>
    <h4 style={styles.title}>{p.title}</h4>
    <p style={styles.sub}>{p.subtitle}</p>
    <p style={styles.purpose}><b>ÇÖZÜMÜN İŞLEVİ</b><br />{p.purposeHint}</p>
    {board}
    {p.hint && <><button style={styles.ghost} onClick={useHint}>💡 İpucu iste</button>{hintOpen && <p style={styles.hint}>{p.hint}</p>}</>}
    <button style={styles.submit} onClick={submit}>{p.interaction?.submitLabel ?? "Kaydı doğrula"}</button>
    {result === "ok" && <p style={styles.ok}><b>IZGARA ETKİSİ</b><br />{p.resultText}</p>}
    {result === "no" && <p style={styles.no}>{p.interaction?.failureMessage ?? "Kayıtlar aynı düzeni vermiyor. Ayrıntıları yeniden karşılaştır."}</p>}
  </section>;
}
