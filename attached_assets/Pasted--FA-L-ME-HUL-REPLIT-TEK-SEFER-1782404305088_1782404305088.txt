// src/components/caseMechanics/FrameShadowProfileMiniGame.jsx
import { useMemo, useState } from "react";

export default function FrameShadowProfileMiniGame({ clue, suspects = [], onSolved }) {
  const [selectedId, setSelectedId] = useState(null);
  const [result, setResult] = useState(null);
  const puzzle = clue?.profilSenteziVerisi ?? {};
  const presentation = puzzle.presentation ?? {};
  const evidence = Array.isArray(puzzle.delilKartlari) ? puzzle.delilKartlari : [];
  const optionIds = Array.isArray(puzzle.optionSuspectIds) ? puzzle.optionSuspectIds : [];
  const candidates = useMemo(() => optionIds.map((id) => suspects.find((s) => s.id === id)).filter(Boolean), [optionIds, suspects]);
  const valid = evidence.length >= 2 && candidates.length >= 2 && Boolean(puzzle.answerSuspectId);

  const select = (suspectId) => {
    setSelectedId(suspectId);
    const ok = suspectId === puzzle.answerSuspectId;
    setResult({
      ok,
      message: ok
        ? (puzzle.successText ?? "Fiziksel işaretler tek bir profilde birleşti.")
        : (puzzle.failureText ?? "Bu profil bütün fiziksel işaretleri aynı anda açıklamıyor."),
    });
    if (ok) onSolved?.({ clueId: clue.id, mechanic: "frame_shadow_profile", suspectId });
  };

  if (!valid) return <p role="alert">Bu vaka için çerçeve profili verisi eksik veya tutarsız.</p>;

  const shell = { marginTop:12, padding:14, border:"1px solid #6d5b34", borderRadius:12, background:"linear-gradient(180deg,#11131a,#0b0e14)", color:"#e8eaf2" };
  const stage = { display:"grid", gridTemplateColumns:"92px minmax(0,1fr)", gap:12, alignItems:"center", margin:"10px 0 12px", padding:11, border:"1px solid #5d4e2e", borderRadius:12, background:"#15130f" };
  const evidenceCard = { border:"1px solid #5a5135", borderRadius:10, padding:10, background:"#171820", color:"#d6dae8" };

  return (
    <section aria-label={presentation.sceneLabel ?? "Çerçeve gölgesi profili"} style={shell}>
      <h4 style={{ margin:"0 0 6px", color:"#f0a830", fontSize:13 }}>🧩 {puzzle.baslik ?? "Çerçeve Gölgesinin Profili"}</h4>
      <div style={stage}>
        <div aria-hidden="true" style={{ width:70, height:88, justifySelf:"center", borderRadius:"36px 36px 12px 12px", background:"radial-gradient(circle at 50% 28%,#0b0d14 0 18px,transparent 19px),linear-gradient(180deg,#31374a,#0d1018 62%)", boxShadow:"inset 0 0 0 2px rgba(224,181,78,.24)" }} />
        <div>
          <div style={{ color:"#e0b54e", fontSize:10, letterSpacing:1.6, fontWeight:800 }}>{presentation.sceneLabel ?? "YANSIMA KAYDI"}</div>
          <p style={{ margin:"5px 0 0", color:"#d7d0b6", fontSize:13, lineHeight:1.5 }}>{presentation.reflectionNote ?? "Yüz seçilemiyor; yüzeyde kalan gerçek işaretleri karşılaştır."}</p>
        </div>
      </div>
      {presentation.purposeHint ? <p style={{ margin:"0 0 12px", padding:10, borderLeft:"2px solid #e0b54e", background:"rgba(224,181,78,.06)", color:"#ecd99a", fontSize:13 }}><b>ÇÖZÜMÜN İŞLEVİ</b><br />{presentation.purposeHint}</p> : null}
      <p style={{ margin:"0 0 9px", color:"#8b91ad", fontSize:13 }}>{puzzle.aciklama}</p>
      <div style={{ color:"#f0a830", fontSize:10, fontWeight:800, letterSpacing:1.4, marginBottom:7 }}>{presentation.physicalEvidenceLabel ?? "FİZİKSEL İŞARETLER"}</div>
      <div style={{ display:"grid", gridTemplateColumns:"repeat(auto-fit,minmax(155px,1fr))", gap:8 }}>
        {evidence.map((card, index) => <article key={index} style={evidenceCard}><b style={{ display:"block", color:"#e0b54e", fontSize:10, letterSpacing:.7, marginBottom:4 }}>{card.baslik ?? card.title ?? "İŞARET"}</b><span style={{ fontSize:12, lineHeight:1.45 }}>{card.metin ?? card.text}</span></article>)}
      </div>
      <div style={{ marginTop:12, color:"#c084fc", fontSize:10, letterSpacing:1.2 }}>PROFİLİ SEÇ — ŞÜPHELİ KARTLARINI İNCELE</div>
      <div style={{ display:"grid", gridTemplateColumns:"repeat(auto-fit,minmax(150px,1fr))", gap:8, marginTop:7 }}>
        {candidates.map((suspect) => {
          const isSelected = selectedId === suspect.id;
          return <button key={suspect.id} type="button" onClick={() => select(suspect.id)} style={{ border:"1px solid " + (isSelected ? (result?.ok ? "#22c55e" : "#dc2626") : "#4d483b"), borderRadius:10, padding:10, textAlign:"left", cursor:"pointer", color:"#e8eaf2", background:isSelected ? (result?.ok ? "rgba(34,197,94,.12)" : "rgba(220,38,38,.10)") : "#151821" }}>{suspect.name}</button>;
        })}
      </div>
      {result ? <p role="status" style={{ margin:"10px 0 0", padding:10, borderRadius:8, background:result.ok ? "rgba(34,197,94,.12)" : "rgba(220,38,38,.12)", color:result.ok ? "#86efac" : "#fca5a5" }}>{result.ok ? "✓ " : "✕ "}{result.message}</p> : null}
    </section>
  );
}
