  const [choices,setChoices]=useState({}); const [order,setOrder]=useState([]);
  const [discs,setDiscs]=useState(()=>Object.fromEntries((p.discs??[]).map(d=>[d.id,d.initial??d.symbols?.[0]??""])));

  if(!styles.has(style))return null;
