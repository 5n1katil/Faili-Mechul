// src/components/caseMechanics/TornRouteReconstructionMiniGame.jsx
import { useMemo, useState } from "react";

export default function TornRouteReconstructionMiniGame({ clue, onSolved, onRequestHint }) {
  const [selected, setSelected] = useState([]);
  const [result, setResult] = useState(null);
  const [hintOpen, setHintOpen] = useState(false);
  const [hintUsed, setHintUsed] = useState(false);
  const sifre = clue?.sifre ?? {};
  const p = sifre.presentation ?? {};
  const pieces = Array.isArray(p.pieces) ? p.pieces : [];
  const correctOrder = Array.isArray(p.correctOrder) ? p.correctOrder : [];
  const interaction = p.interaction ?? {};
  const pieceById = useMemo(() => new Map(pieces.map((piece) => [piece.id, piece])), [pieces]);
  const valid = pieces.length >= 4 && correctOrder.length >= 3 && correctOrder.every((id) => pieceById.has(id));
  const max = correctOrder.length;
  const toggle = (id) => {
    setResult(null);
    setSelected((before) => before.includes(id)
      ? before.filter((value) => value !== id)
      : (before.length < max ? [...before, id] : before));
  };
  const reset = () => { setSelected([]); setResult(null); };
  const submit = () => {
    const ok = selected.length === correctOrder.length && selected.every((id, index) => id === correctOrder[index]);
    setResult({ ok, message: ok ? (interaction.successMessage ?? "Rota doğru kuruldu.") : (interaction.failureMessage ?? "Yırtıkları ve rota sözcüklerini yeniden kontrol et.") });
    if (ok) onSolved?.({ clueId: clue.id, mechanic: "torn_route_reconstruction", order: selected });
  };
  const openHint = () => {
    if (!hintUsed) {
      onRequestHint?.({ clueId: clue.id, penaltySeconds: Number(sifre.hintPenaltySeconds ?? 60) });
      setHintUsed(true);
    }
    setHintOpen(true);
  };
  if (!valid) return <div role="alert">Bu vaka için yırtık rota parçaları eksik veya tutarsız.</div>;

  const shell = { background:"#10131f", border:"1px solid #262c44", borderRadius:12, padding:14, color:"#e8eaf2" };
  const pieceStyle = { position:"relative", minHeight:105, border:"1px solid #5a5135", borderRadius:10, background:"#23251c", color:"#f4eccb", padding:10, textAlign:"left", cursor:"pointer" };

  return (
    <section style={shell} aria-label={p.accessibility?.description ?? "Yırtık rota sayfası"}>
      <h4 style={{ margin:"0 0 6px", color:"#f0a830" }}>{p.title ?? "Yırtık Rota Sayfası"}</h4>
      <p style={{ margin:"0 0 10px", color:"#8b91ad", fontSize:13 }}>{p.subtitle ?? "Parçaları sağ ve sol yırtık izlerine göre sırala."}</p>
      {p.purposeHint ? <p style={{ margin:"0 0 12px", padding:10, borderLeft:"2px solid #e0b54e", background:"rgba(224,181,78,.06)", color:"#ecd99a", fontSize:13 }}><b>ÇÖZÜMÜN İŞLEVİ</b><br/>{p.purposeHint}</p> : null}

      <div style={{ display:"grid", gridTemplateColumns:"repeat(auto-fit,minmax(170px,1fr))", gap:9 }}>
        {pieces.map((piece) => {
          const pos = selected.indexOf(piece.id);
          return <button key={piece.id} type="button" onClick={() => toggle(piece.id)} aria-pressed={pos >= 0} style={{ ...pieceStyle, borderColor:pos >= 0 ? "#22c55e" : "#5a5135", background:pos >= 0 ? "#213426" : "#23251c" }}>
            <span style={{ display:"flex", justifyContent:"space-between", gap:8, fontSize:10, color:"#cdbb79" }}><span>◁ {piece.leftTear}</span><span>{piece.label}</span><span>{piece.rightTear} ▷</span></span>
            <span style={{ display:"block", marginTop:16, fontFamily:"serif", fontWeight:700, lineHeight:1.35 }}>{piece.text}</span>
            {pos >= 0 ? <span style={{ position:"absolute", right:8, bottom:7, width:20, height:20, borderRadius:"50%", background:"#22c55e", color:"#07110a", display:"grid", placeItems:"center", fontWeight:900 }}>{pos + 1}</span> : null}
          </button>;
        })}
      </div>

      <div style={{ marginTop:12, padding:10, border:"1px dashed #655d45", borderRadius:10, background:"#10120e" }}>
        <b style={{ fontSize:10, letterSpacing:1.2, color:"#f0a830" }}>SEÇİLEN ROTA SIRASI</b>
        <div style={{ display:"flex", gap:6, flexWrap:"wrap", marginTop:8 }}>
          {selected.length ? selected.map((id, index) => <span key={id} style={{ fontSize:12, color:"#d6dae8" }}>{index ? " → " : ""}{pieceById.get(id)?.label}</span>) : <span style={{ fontSize:12, color:"#8b91ad" }}>Önce yırtık parçaları seç.</span>}
        </div>
      </div>

      <div style={{ display:"flex", gap:8, flexWrap:"wrap", marginTop:10 }}>
        <button type="button" onClick={reset} style={{ flex:1, padding:10, borderRadius:8, border:"1px solid #262c44", background:"#1c2138", color:"#e8eaf2" }}>{interaction.resetLabel ?? "Dizilişi Temizle"}</button>
        <button type="button" onClick={submit} style={{ flex:1, padding:10, border:0, borderRadius:8, background:"#e0b54e", color:"#1a1205", fontWeight:700 }}>{interaction.submitLabel ?? "Rotayı Doğrula"}</button>
      </div>

      <button type="button" onClick={openHint} disabled={hintUsed} style={{ width:"100%", marginTop:10, padding:10, borderRadius:8, border:"1px dashed #b45309", background:"transparent", color:"#fbbf24" }}>{hintUsed ? "İpucu açıldı" : "İpucu iste (-" + Number(sifre.hintPenaltySeconds ?? 60) + " sn)"}</button>
      {hintOpen ? <p style={{ margin:"8px 0 0", padding:9, borderLeft:"2px solid #b45309", background:"rgba(180,83,9,.08)", color:"#fbbf24", fontSize:13 }}>{sifre.cozumIpucu}</p> : null}
      {result ? <p role="status" style={{ margin:"10px 0 0", padding:10, borderRadius:8, background:result.ok ? "rgba(34,197,94,.12)" : "rgba(220,38,38,.12)", color:result.ok ? "#86efac" : "#fca5a5" }}>{result.ok ? "✓ " : "✕ "}{result.message}</p> : null}
      {result?.ok ? <>
        <p style={{ margin:"10px 0 0", padding:10, border:"1px solid #22c55e", borderRadius:8, background:"rgba(34,197,94,.08)", color:"#b7f7c8" }}><b>TAMAMLANAN ROTA</b><br/>{p.routeResult}</p>
        {sifre.aciklama ? <p style={{ margin:"10px 0 0", color:"#8b91ad", fontStyle:"italic", fontSize:13 }}>{sifre.aciklama}</p> : null}
      </> : null}
    </section>
  );
}
